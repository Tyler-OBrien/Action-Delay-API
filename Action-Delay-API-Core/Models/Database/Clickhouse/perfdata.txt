CREATE TABLE job_runs_perf
(
 `job_name` LowCardinality(String),
 `run_time` DateTime CODEC(Delta, ZSTD),
 `run_status` LowCardinality(String),
 `average_response_latency` UInt32,
 `average_binding_response_latency` UInt32,
 `outcome` LowCardinality(String) DEFAULT ''
)
ENGINE = MergeTree
PRIMARY KEY (run_time, job_name)
PARTITION BY toYYYYMM(run_time)
ORDER BY (run_time, job_name);

CREATE TABLE api_errors_perf
(
 `job_name` LowCardinality(String),
 `location_name` LowCardinality(String),
 `run_time` DateTime CODEC(Delta, ZSTD),
 `error_type` LowCardinality(String),
 `error_description` String,
 `error_hash` String, 
 `response_latency` UInt32,
 `location_id` LowCardinality(String),
)
ENGINE = MergeTree
PRIMARY KEY (run_time, error_hash)
PARTITION BY toYYYYMM(run_time)
ORDER BY (run_time, error_hash);



CREATE TABLE job_runs_locations_perf
(
 `job_name` LowCardinality(String),
 `location_name` LowCardinality(String),
 `run_time` DateTime CODEC(Delta, ZSTD),
 `run_status` LowCardinality(String),
 `response_latency` UInt32,
 `location_id` LowCardinality(String),
 `binding_response_latency` UInt32,
 `outcome` LowCardinality(String) DEFAULT '',
 `failure_reason` LowCardinality(String) DEFAULT ''
)
ENGINE = MergeTree
PRIMARY KEY (run_time, job_name, location_name)
PARTITION BY toYYYYMM(run_time)
ORDER BY (run_time, job_name, location_name);

CREATE MATERIALIZED VIEW job_runs_perf_mv_30_mins
ENGINE = AggregatingMergeTree
ORDER BY (job_name, average_time)
POPULATE
AS SELECT
    job_name,
    average_time,
    minState(average_response_latency) as min_average_response_latency,
    maxState(average_response_latency) as max_average_response_latency,
    avgState(average_response_latency) as avg_average_response_latency,
    quantilesState(0.5, 0.9, 0.99)(average_response_latency) as quan_average_response_latency
FROM job_runs_perf
where run_status = 'Success'
GROUP BY job_name, toStartOfInterval(run_time, INTERVAL 30 minute) as average_time;

-- Target table for colo-level 5-minute pre-aggregated perf data (30-day TTL)
CREATE TABLE job_runs_locations_perf_colo_5m
(
    job_name        LowCardinality(String),
    location_name   LowCardinality(String),
    location_id     LowCardinality(String),
    time_bucket     DateTime CODEC(Delta, ZSTD),
    run_count       UInt64,
    sum_latency     UInt64,
    median_latency  AggregateFunction(quantile(0.5), UInt32)
)
ENGINE = AggregatingMergeTree()
PARTITION BY toYYYYMM(time_bucket)
ORDER BY (job_name, location_name, location_id, time_bucket)
TTL time_bucket + INTERVAL 30 DAY;

-- Materialized view that populates job_runs_locations_perf_colo_5m
CREATE MATERIALIZED VIEW job_runs_locations_perf_colo_5m_mv
TO job_runs_locations_perf_colo_5m
AS SELECT
    job_name,
    location_name,
    location_id,
    toStartOfFiveMinutes(run_time)       AS time_bucket,
    count()                              AS run_count,
    sum(response_latency)                AS sum_latency,
    quantileState(0.5)(response_latency) AS median_latency
FROM job_runs_locations_perf
WHERE job_name IN (
    'cloudflare-worker-uncached-10kb-free-plan',
    'cloudflare-worker-uncached-10kb-pro-plan',
    'cloudflare-worker-uncached-10kb-business-plan',
    'cloudflare-worker-uncached-10kb',
    'cloudflare-worker-uncached-10kb-argo',
    'cloudflare-worker-uncached-10kb-spectrum'
)
GROUP BY job_name, location_name, location_id, time_bucket;
